   908	/// One supervisor per device: it restarts the daemon pair while Sui stays enabled and exits
   909	/// as soon as Sui is disabled, so it never outlives the extension it watches.
   910	fn watchdog() -> Result<()> {
   911	    let lock = OpenOptions::new()
   912	        .read(true)
   913	        .write(true)
   914	        .create(true)
   915	        .truncate(false)
   916	        .mode(0o600)
   917	        .custom_flags(libc::O_NOFOLLOW)
   918	        .open(WATCHDOG_LOCK)?;
   919	    ensure!(
   920	        unsafe { libc::flock(lock.as_raw_fd(), libc::LOCK_EX | libc::LOCK_NB) } == 0,
   921	        "Another Sui watchdog is already running"
   922	    );
   923	    let mut child: Option<Child> = None;
   924	    let mut started = Instant::now();
   925	    let mut refreshed: Option<Instant> = None;
   926	    loop {
   927	        if !active()? {
   928	            if let Some(mut child) = child {
   929	                let _ = child.kill();
   930	                let _ = child.wait();
   931	            }
   932	            return Ok(());
   933	        }
   934	        if boot_completed()
   935	            && refreshed.is_none_or(|time| time.elapsed() >= Duration::from_secs(30))
   936	        {
   937	            if let Err(error) = register_manager() {
   938	                log::warn!("Sui manager refresh failed: {error:#}");
   939	            }
   940	            refreshed = Some(Instant::now());
   941	        }
   942	        if let Some(process) = child.as_mut()
   943	            && process.try_wait()?.is_some()
   944	        {
   945	            child = None;
   946	        }
   947	        if daemon_healthy() {
   948	            if !boot_marker_matches() {
   949	                write_boot_marker()?;
   950	            }
   951	        } else if (child.is_none()
   952	            || (boot_completed() && started.elapsed() >= DAEMON_READY_TIMEOUT))
   953	            && let Ok(_operation) = self::lock()
   954	            && active()?
   955	        {
   956	            stop_daemon();
   957	            if let Some(mut previous) = child.take() {
   958	                let _ = previous.kill();
   959	                let _ = previous.wait();
   960	            }
   961	            match start_daemon() {
   962	                Ok(process) => {
   963	                    child = Some(process);
   964	                    started = Instant::now();
   965	                }
   966	                Err(error) => log::warn!("Sui watchdog restart failed: {error:#}"),
   967	            }
   968	        }
   969	        std::thread::sleep(Duration::from_secs(1));
   970	    }
   971	}

   973	fn spawn_watchdog() -> Result<()> {
   974	    let mut command = Process::new("/data/adb/ksud");
   975	    command
   976	        .args(["sui", "watchdog"])
   977	        .stdin(Stdio::null())
   978	        .stdout(Stdio::null())
   979	        .stderr(Stdio::null());
   980	    unsafe {
   981	        command.pre_exec(|| {
   982	            leave_service_cgroup()?;
   983	            libc::setsid();
   984	            Ok(())
   985	        });
   986	    }
   987	    let mut child = command.spawn().context("Start the Sui watchdog")?;
   988	    std::thread::spawn(move || {
   989	        let _ = child.wait();
   990	    });

    74	pub fn disable() -> Result<()> {
    75	    // Only the flag goes; the monitor and its daemons keep running so the current
    76	    // session is not disturbed. The module system behaves the same way: a disable file
    77	    // takes effect on the next boot, it does not kill a running Zygisk. They are
    78	    // stopped by a soft reboot and die with the system on a real one, at which point
    79	    // `cleanup_disabled` drops the files.
    80	    let flag = Path::new(BUILTIN_ZYGISK_FLAG);
    81	    if flag.exists() {
    82	        fs::remove_file(flag).with_context(|| "Failed to remove builtin zygisk flag")?;
    83	    }
    84	    info!("Built-in Zygisk disabled, its processes stop at the next reboot");
    85	    Ok(())
    86	}

    89	///
    90	/// Only safe with no live monitor: a boot without the flag reaches this after the reboot
    91	/// killed the previous session's processes, and a soft reboot calls it after `shutdown`.
    92	/// The work dir is shared with a standalone NyaZygisk module, which keeps its options and
    93	/// reported status there; it is only dropped when something of ours was staged next to it
    94	/// and no standalone NyaZygisk stands in for the built-in.
    95	pub fn cleanup_disabled() -> Result<()> {
    96	    if is_enabled() {
    97	        return Ok(());

   103	        warn!("Built-in Zygisk cleanup skipped, its monitor is still running");
   104	        return Ok(());

    56	fun sha256(file: File): String {
    57	    val digest = MessageDigest.getInstance("SHA-256")
    58	    file.inputStream().use { input ->
    59	        val buffer = ByteArray(DEFAULT_BUFFER_SIZE)
    60	        while (true) {
    61	            val bytesRead = input.read(buffer)
    62	            if (bytesRead < 0) break
    63	            digest.update(buffer, 0, bytesRead)
    64	        }
    65	    }
    66	    return digest.digest().joinToString("") { "%02x".format(it) }

   240	        val zipTask = tasks.register<Zip>("zip${variantCapped}") {
   241	            dependsOn("prepareMagiskFiles${variantCapped}")
   242	            from(magiskDir)
   243	            archiveFileName.set(zipName)
   244	            destinationDirectory.set(outDir)
   245	        }
   246	
   247	        fun registerAdbTask(
   248	            name: String,
   249	            installerMode: ModuleInstallerMode? = null,
   250	            shouldReboot: Boolean = false
   251	        ) {
   252	            tasks.register<AdbModuleTask>(name) {
   253	                dependsOn(zipTask)
   254	                moduleZip.set(zipTask.flatMap { it.archiveFile })
   255	                installerMode?.let { installer.set(it) }
   256	                reboot.set(shouldReboot)
   257	                device.convention(providers.gradleProperty("device"))
   258	            }
   259	        }
   260	
   261	        registerAdbTask("push${variantCapped}")
   262	        registerAdbTask("flash${variantCapped}", ModuleInstallerMode.AUTO)
   263	        registerAdbTask("flashWithMagisk${variantCapped}", ModuleInstallerMode.MAGISK)
   264	        registerAdbTask("flashWithKsud${variantCapped}", ModuleInstallerMode.KERNEL_SU)
   265	        registerAdbTask("flashAndReboot${variantCapped}", ModuleInstallerMode.AUTO, shouldReboot = true)
   266	        registerAdbTask(
   267	            "flashWithMagiskAndReboot${variantCapped}",
   268	            ModuleInstallerMode.MAGISK,
   269	            shouldReboot = true
   270	        )
   271	        registerAdbTask(
   272	            "flashWithKsudAndReboot${variantCapped}",
   273	            ModuleInstallerMode.KERNEL_SU,
   274	            shouldReboot = true
   275	        )
