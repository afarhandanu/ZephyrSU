    17	/**
    18	 * Sui is a Zygisk module, so its switch stays disabled until ksud reports a usable provider.
    19	 * The permission UI itself stays Sui's own; this row only owns install, enable and diagnosis.
    20	 */
    21	@Composable
    22	internal fun SuiStatusItem(state: SettingsUiState, actions: SettingsScreenActions) {
    23	    val status = state.suiStatus
    24	    val confirm = rememberConfirmDialog(onConfirm = { actions.onSetSuiEnabled(true) })
    25	    val notInstalled = stringResource(R.string.sui_not_installed)
    26	    val stateText = status?.let { stringResource(suiStateRes(it)) }
    27	    val reasonText = status?.let { suiReasonRes(it)?.let { res -> stringResource(res, it.detail) } }
    28	    val zygiskText = status?.let { stringResource(zygiskStateRes(it)) }.orEmpty()
    29	    val confirmTitle = stringResource(R.string.sui_confirm_title)
    30	    val confirmAccept = stringResource(R.string.sui_confirm_confirm)
    31	    val confirmCancel = stringResource(R.string.sui_confirm_cancel)
    32	    val confirmBody = status?.let {
    33	        stringResource(
    34	            R.string.sui_confirm_body,
    35	            zygiskText,
    36	            it.payloadVersion.ifBlank { notInstalled },
    37	        )
    38	    }
    39	    val summary = when {
    40	        state.isSuiLoading -> stringResource(R.string.sui_loading)
    41	        state.suiReadError || status == null -> stringResource(R.string.sui_read_error)
    42	        else -> listOfNotNull(
    43	            stateText,
    44	            reasonText,
    45	            stringResource(R.string.sui_dependency, zygiskText),
    46	        ).joinToString("\n")
    47	    }
    48	    val actionable = status != null && !state.suiReadError && !state.isSuiLoading && !state.isSuiWriting
    49	
    50	    SegmentedSwitchItem(
    51	        icon = Icons.Filled.WaterDrop,
    52	        title = stringResource(R.string.sui_title),
    53	        summary = summary,
    54	        checked = status?.selected == true,
    55	        enabled = actionable && status.conflicts.isEmpty(),
    56	        onCheckedChange = { wanted ->
    57	            if (!wanted) {
    58	                actions.onSetSuiEnabled(false)
    59	            } else {
    60	                confirm.showConfirm(
    61	                    title = confirmTitle,
    62	                    content = confirmBody,
    63	                    confirm = confirmAccept,
    64	                    dismiss = confirmCancel,
    65	                )
    66	            }
    67	        },
    68	    )
    69	
    70	    if (status != null && status.installed) {
    71	        SegmentedListItem(
    72	            headlineContent = { Text(stringResource(R.string.sui_manage)) },
    73	            supportingContent = { Text(stringResource(R.string.sui_manage_summary)) },
    74	            leadingContent = { Icon(Icons.AutoMirrored.Filled.OpenInNew, null) },
    75	            enabled = !state.isSuiWriting,
    76	            onClick = actions.onManageSuiPermissions,
    77	        )
    78	    }
    79	
    80	    if (status != null && status.needsDiagnostics) {
    81	        SuiDiagnostics(
    82	            status = status,
    83	            onRepair = actions.onRepairSui,
    84	            busy = state.isSuiWriting,
    85	            managementAvailable = state.suiManagementAvailable,
    86	        )
    87	    }
    88	}

   152	@StringRes
   153	private fun suiStateRes(status: SuiStatus): Int = when (status.state) {
   154	    SuiStatus.STATE_ENABLED -> R.string.sui_state_enabled
   155	    SuiStatus.STATE_PENDING_REBOOT -> R.string.sui_state_pending
   156	    SuiStatus.STATE_EXTERNAL_UNVERIFIED -> R.string.sui_state_external
   157	    SuiStatus.STATE_UNAVAILABLE -> R.string.sui_state_unavailable
   158	    SuiStatus.STATE_ERROR -> R.string.sui_state_error
   159	    else -> R.string.sui_state_disabled
   160	}

   162	@StringRes
   163	private fun suiReasonRes(status: SuiStatus): Int? = when (status.reason) {
   164	    "zygisk_disabled" -> R.string.sui_reason_zygisk_disabled
   165	    "zygisk_engine_missing" -> R.string.sui_reason_zygisk_engine_missing
   166	    "zygisk_startup_protection" -> R.string.sui_reason_zygisk_startup
   167	    "zygisk_injection_failed" -> R.string.sui_reason_injection_failed
   168	    "conflict_existing_sui" -> R.string.sui_reason_conflict
   169	    "payload_missing" -> R.string.sui_reason_payload_missing
   170	    "module_disabled" -> R.string.sui_reason_module_disabled
   171	    "enabled_externally" -> R.string.sui_reason_enabled_externally
   172	    "not_installed" -> null
   173	    else -> R.string.sui_reason_unknown.takeIf { status.state == SuiStatus.STATE_ERROR }
   174	}

   176	@StringRes
   177	private fun zygiskStateRes(status: SuiStatus): Int = when {
   178	    status.externalProviders.isNotEmpty() -> R.string.sui_zygisk_external
