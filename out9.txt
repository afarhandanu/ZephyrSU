   284	    let mut abi: Option<String> = None;
   285	    let mut found = Vec::new();

   376	/// One `/proc` entry, read once and matched against every module.
   377	struct ProcessEntry {
   378	    pid: i32,
   379	    /// `argv[0]`, which is what companions and targets are identified by.
   380	    argv0: String,
   381	    argv: Vec<String>,
   382	    exe: Option<PathBuf>,
   383	}
   384	
   385	fn read_process(pid: i32) -> Option<ProcessEntry> {
   386	    let raw = fs::read(format!("/proc/{pid}/cmdline")).ok()?;
   387	    if raw.is_empty() {
   388	        return None;
   389	    }
   390	    let mut argv: Vec<String> = raw
   391	        .split(|byte| *byte == 0)
   392	        .filter(|part| !part.is_empty())
   393	        .map(|part| String::from_utf8_lossy(part).into_owned())
   394	        .collect();
   395	    if argv.is_empty() {
   396	        return None;
   397	    }
   398	    let argv0 = argv.remove(0);
   399	    Some(ProcessEntry {
   400	        pid,
   401	        argv0,
   402	        argv,
   403	        exe: fs::read_link(format!("/proc/{pid}/exe")).ok(),
   404	    })
   405	}
   406	
   407	fn scan_processes() -> Vec<ProcessEntry> {
   408	    let Ok(entries) = fs::read_dir("/proc") else {
   409	        return Vec::new();
   410	    };
   411	    let mut processes: Vec<ProcessEntry> = entries
   412	        .flatten()
   413	        .filter_map(|entry| entry.file_name().to_str()?.parse::<i32>().ok())
   414	        .filter_map(read_process)
   415	        .collect();
   416	    processes.sort_by_key(|process| process.pid);
   417	    processes
   418	}

   420	/// Whether `process` is this module's companion.
   421	fn is_companion(process: &ProcessEntry, id: &str) -> bool {
   422	    let Some(name) = companion_name(&process.argv0) else {
   423	        return false;
   424	    };
   425	    if name == "zn-companion" {
   426	        // `zygiskd-zn-companion zn-companion <lib> <fd>`: the library names the module.
   427	        return process
   428	            .argv
   429	            .get(1)
   430	            .is_some_and(|lib| lib.starts_with(&format!("/data/adb/modules/{id}/")));
   431	    }
   432	    name == id

   435	/// Whether `process` is one of the injection targets the module declared.
   436	///
   437	/// Targets are stored without their `name=` / `path=` prefix: an absolute path matches
   438	/// the executable, anything else the process name.
   439	fn matches_target(process: &ProcessEntry, target: &str) -> bool {
   440	    if target.starts_with('/') {
   441	        return process
   442	            .exe
   443	            .as_deref()
   444	            .is_some_and(|exe| exe == Path::new(target));
   445	    }
   446	    let argv0 = process.argv0.rsplit('/').next().unwrap_or("");
   447	    argv0 == target
