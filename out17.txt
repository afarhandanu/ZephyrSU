    88	        match fs::read(STATE) {
    89	            Ok(bytes) => serde_json::from_slice(&bytes).context("Invalid Sui state"),
    90	            Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(Self::default()),
    91	            Err(error) => Err(error).context("Read Sui state"),
    92	        }
    93	    }

    95	    fn save(&self) -> Result<()> {
    96	        fs::create_dir_all(STATE_DIR)?;
    97	        let mut file = tempfile::NamedTempFile::new_in(STATE_DIR)?;
    98	        serde_json::to_writer(&mut file, self).context("Serialize Sui state")?;
    99	        file.write_all(b"\n")?;
   100	        file.persist(STATE).context("Persist Sui state")?;
   101	        Ok(())
   102	    }
   103	}
