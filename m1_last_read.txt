const MARKER_PATH: &str = concatcp!(crate::defs::WORKING_DIR, ".boot_hide");

#[derive(Debug, Clone, Subcommand)]
pub enum Command {
    Status,
    Enable,
    Disable,
}

fn apply_with(
    mut get: impl FnMut(&str) -> Option<String>,
    mut set: impl FnMut(&str, &str) -> bool,
    mut delete: impl FnMut(&str),
) -> usize {
    let mut writes = 0;
    for &(key, value) in &PROPERTIES {
        let Some(current) = get(key) else {
            continue;
        };
        if current != value {
            set(key, value);
            writes += 1;
        }
    }
    if get("service.adb.root").is_some() {
        delete("service.adb.root");
    }
    writes
}

#[cfg(target_os = "android")]
fn marker_path() -> &'static Path {
    Path::new(MARKER_PATH)
}

#[cfg(target_os = "android")]
fn marker_exists() -> Result<bool> {
    marker_path()
        .try_exists()
        .context("Inspect FinaSU Hide marker")
}

#[cfg(target_os = "android")]
fn write_marker() -> Result<()> {
    let path = marker_path();
    let mut file = OpenOptions::new()
        .write(true)
        .create(true)
        .truncate(true)
        .mode(0o600)
        .custom_flags(libc::O_NOFOLLOW)
        .open(path)
        .with_context(|| format!("Open {}", path.display()))?;
    file.set_permissions(Permissions::from_mode(0o600))
        .context("Set FinaSU Hide marker permissions")?;
    file.flush().context("Write FinaSU Hide marker")?;
    file.sync_all().context("Sync FinaSU Hide marker")?;
    File::open(crate::defs::WORKING_DIR)
        .context("Open working directory")?
        .sync_all()
        .context("Sync working directory")?;
    Ok(())
}

#[cfg(target_os = "android")]
fn remove_marker() -> Result<()> {
    match std::fs::remove_file(marker_path()) {
        Ok(()) => {}
        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(()),
        Err(error) => return Err(error).context("Remove FinaSU Hide marker"),
    }
    File::open(crate::defs::WORKING_DIR)
        .context("Open working directory")?
        .sync_all()
        .context("Sync working directory")?;
    Ok(())
}

#[cfg(target_os = "android")]
fn rebuild_if_needed(rp: &ResetProp, key: &str, needed: bool) -> Result<()> {
    if needed {
        let context = sys_prop::get_context(key)
            .with_context(|| format!("Get property context for {key}"))?;
        rp.rebuild(&context)
            .with_context(|| format!("Rebuild property area for {key}"))?;
    }
    Ok(())
}

#[cfg(target_os = "android")]
fn apply() -> Result<usize> {
    sys_prop::init().context("Initialize system property API")?;
    let rp = ResetProp {
        skip_svc: true,
        persistent: false,
        persist_only: false,
        verbose: false,
        show_context: false,
        rebuild: false,
    };

    let mut set_error = None;
    let mut delete_error = None;
    let writes = apply_with(
        sys_prop::get,
        |key, value| {
            if set_error.is_some() {
                return false;
            }
            match rp
                .set(key, value)
                .with_context(|| format!("Set {key}"))
                .and_then(|needed| rebuild_if_needed(&rp, key, needed))
            {
                Ok(()) => true,
                Err(error) => {
                    set_error = Some(error);
                    false
                }
            }
        },
        |key| {
            if delete_error.is_some() {
                return;
            }
            if let Err(error) = rp
                .delete(key)
                .with_context(|| format!("Delete {key}"))
                .and_then(|needed| rebuild_if_needed(&rp, key, needed))
            {
                delete_error = Some(error);
            }
        },
    );

    if let Some(error) = set_error.or(delete_error) {
        return Err(error);
    }
    Ok(writes)
}

#[cfg(target_os = "android")]
fn ensure_root() -> Result<()> {
    if unsafe { libc::geteuid() } != 0 {
        bail!("Root is required for FinaSU Hide");
    }
    Ok(())
}

#[cfg(target_os = "android")]
pub fn boot_apply() {
    match marker_exists() {
        Ok(false) => return,
        Ok(true) => {}
        Err(error) => {
            warn!("Inspect FinaSU Hide marker failed: {error:#}");
            return;
        }
    }
    if crate::utils::is_safe_mode() {
        warn!("safe mode, skip FinaSU Hide");
        return;
    }
    match apply() {
        Ok(writes) => info!("FinaSU Hide applied {writes} properties"),
        Err(error) => warn!("Apply FinaSU Hide failed: {error:#}"),
    }
}

#[cfg(target_os = "android")]
pub fn run(command: &Command) -> Result<()> {
    ensure_root()?;
    match command {
        Command::Status => {
            println!(
                "{}",
                if marker_exists()? {
                    "enabled"
                } else {
                    "disabled"
                }
            );
            Ok(())
        }
        Command::Enable => {
            write_marker()?;
            if let Err(error) = apply() {
                warn!("Apply FinaSU Hide failed: {error:#}");
            }
            Ok(())
        }
        Command::Disable => remove_marker(),
    }
}

 * summaries and 14sp captions over generous line boxes. A tighter scale read as too small on a
 * large-screened device, while the Material page sizes read as too loose for a settings-style
 * list. Each line box keeps the same gap over its text, so only the size moves.
 */

        @Composable get() = scale(size = 17.sp, lineHeight = 22.sp, weight = FontWeight.Medium, lineBreak = LineBreak.Heading)

    val Summary: TextStyle

    /** The small print: 14sp over an 18sp line box. */
    val Caption: TextStyle
        @Composable get() = scale(size = 14.sp, lineHeight = 18.sp)

     * [Caption] for machine strings - version numbers, build ids, fingerprints. These are read as
     * whole tokens, so hyphenation is off and the line breaker balances and breaks between phrases
     * instead of filling the first line and stranding a character on the second.

        @Composable get() = Caption.copy(
            hyphens = Hyphens.None,
            lineBreak = LineBreak.Heading,

}

/** Complete the copy before callers publish a new filename or remove the previous resource. */
internal object MediaFiles {
    suspend fun import(context: Context, uri: Uri, directory: File, stem: String, fallback: String): File =
        withContext(Dispatchers.IO) {
            check(directory.isDirectory || directory.mkdirs())
            val extension = MimeTypeMap.getSingleton()
                .getExtensionFromMimeType(context.contentResolver.getType(uri)) ?: fallback
            val target = File.createTempFile("${stem}_", ".$extension", directory)
            try {
                requireNotNull(context.contentResolver.openInputStream(uri)).use { input ->
                    target.outputStream().use { output -> input.copyTo(output) }
                }
                require(target.length() > 0) { "Empty media file" }
                target
            } catch (error: Exception) {
                target.delete()
                throw error
            }
        }

package me.weishu.kernelsu.media

import android.content.Context
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.mutableFloatStateOf

object VibrationConfig {
    private const val PREFS_NAME = "vibration_settings"
    private const val KEY_ENABLED = "vibration_enabled"
    private const val KEY_INTENSITY = "vibration_intensity"
    private const val KEY_SCOPE = "vibration_scope"

    const val SCOPE_GLOBAL = "global"
    const val SCOPE_BOTTOM_BAR = "bottom_bar"

    var isVibrationEnabled: Boolean by mutableStateOf(false)
        private set

    var vibrationIntensity: Float by mutableFloatStateOf(0.5f)
        private set

    var scope: String by mutableStateOf(SCOPE_GLOBAL)
        private set

    fun setEnabledState(enabled: Boolean) {
        isVibrationEnabled = enabled
    }

    fun setIntensityValue(value: Float) {
        vibrationIntensity = value.coerceIn(0f, 1f)
    }

    fun setScopeValue(value: String) {
        scope = if (value == SCOPE_BOTTOM_BAR) value else SCOPE_GLOBAL
    }

    fun load(context: Context) {
        val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
        isVibrationEnabled = prefs.getBoolean(KEY_ENABLED, false)
        setIntensityValue(prefs.getFloat(KEY_INTENSITY, 0.5f))
        setScopeValue(prefs.getString(KEY_SCOPE, SCOPE_GLOBAL) ?: SCOPE_GLOBAL)
    }

    fun save(context: Context) {
        val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
        prefs.edit()
            .putBoolean(KEY_ENABLED, isVibrationEnabled)
            .putFloat(KEY_INTENSITY, vibrationIntensity)
            .putString(KEY_SCOPE, scope)
            .apply()
    }
}

package me.weishu.kernelsu.media

import android.content.Context
import android.os.VibrationEffect
import android.os.VibratorManager

object VibrationManager {

    fun vibrate(context: Context) {
        if (!VibrationConfig.isVibrationEnabled) return
        if (VibrationConfig.vibrationIntensity <= 0f) return

        try {
            val vibrator = context.getSystemService(VibratorManager::class.java)?.defaultVibrator ?: return

            if (vibrator.hasVibrator()) {
                // Map 0.0-1.0 to 1-255
                val intensity = (VibrationConfig.vibrationIntensity * 255).toInt().coerceIn(1, 255)
                // A short duration for a "tick" or "click" feel
                val duration = 30L

                val amplitude = if (vibrator.hasAmplitudeControl()) intensity else VibrationEffect.DEFAULT_AMPLITUDE
                vibrator.vibrate(VibrationEffect.createOneShot(duration, amplitude))
            }
        } catch (e: Exception) {
            // Catch all exceptions including SecurityException to prevent crash
            android.util.Log.e("VibrationManager", "Failed to vibrate", e)
        }
    }
}

/**
 * A media setting driven by a slider.
 *
 * A drag updates the setting on every frame, so a volume move is heard while the thumb travels;
 * [onCommit] only persists — and plays a preview — once the gesture has settled. The value settles
 * on the whole percent the row shows, so the label always names the value that was stored.
 */
@Composable
internal fun MediaSlider(
    title: String,
    value: Float,
    icon: ImageVector? = null,
    onValueChange: (Float) -> Unit,
    onCommit: () -> Unit,
) {
    SegmentedSliderItem(icon = icon, title = title, value = value, valueRange = 0f..1f,
        valueText = { "${(it * 100).roundToInt()}%" },
        onValueChange = { onValueChange(snapToPercent(it)) },
        onValueChangeFinished = { onCommit() })
}
